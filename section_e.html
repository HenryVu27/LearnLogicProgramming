import math

def cosine_similarity(vec_a: list[float], vec_b: list[float]) -> float:
    if len(vec_a) != len(vec_b):
        raise ValueError("Vectors must have the same length")

    dot = sum(a * b for a, b in zip(vec_a, vec_b))
    mag_a = math.sqrt(sum(a * a for a in vec_a))
    mag_b = math.sqrt(sum(b * b for b in vec_b))

    if mag_a == 0 or mag_b == 0:
        raise ValueError("Zero vector has no direction")

    return dot / (mag_a * mag_b)

Cosine similarity = dot(A, B) / (||A|| * ||B||). It measures the angle between two vectors, not their magnitude. Identical directions = 1, orthogonal = 0, opposite = -1. This is what FAISS computes millions of times per query in FirstThen’s retrieval layer.

def chunk_text(text: str, chunk_size: int, overlap: int) -> list[str]:
    sentences = text.split('. ')
    sentences = [s + '.' if not s.endswith('.') else s for s in sentences if s.strip()]

    chunks = []
    current = ""
    for sent in sentences:
        if len(current) + len(sent) + 1 > chunk_size and current:
            chunks.append(current.strip())
            # Backtrack by overlap characters
            current = current[-overlap:].strip() + " " + sent if overlap > 0 else sent
        else:
            current = (current + " " + sent).strip() if current else sent

    if current.strip():
        chunks.append(current.strip())
    return chunks

Chunking strategy directly affects retrieval quality. Too large = diluted embeddings. Too small = lost context. Overlap ensures sentences at chunk boundaries aren’t orphaned. FirstThen’s clinical knowledge base (ADHD strategies, parenting guides) needs careful chunking so each chunk is a coherent unit of advice.

import math

def bm25_score(query_terms: list[str], document: list[str],
               corpus: list[list[str]], k1: float = 1.5, b: float = 0.75) -> float:
    """
    Compute BM25 score for a single document given a query.

    where N = total docs, n(term) = docs containing term,
    tf = term frequency in document, doc_len = len(document),
    avg_dl = average doc length in corpus
    """

import math

def bm25_score(query_terms, document, corpus, k1=1.5, b=0.75):
    N = len(corpus)
    avg_dl = sum(len(d) for d in corpus) / N
    doc_len = len(document)
    score = 0.0

    for term in query_terms:
        tf = document.count(term)
        if tf == 0:
            continue
        n_term = sum(1 for d in corpus if term in d)
        idf = math.log((N - n_term + 0.5) / (n_term + 0.5) + 1)
        numerator = tf * (k1 + 1)
        denominator = tf + k1 * (1 - b + b * (doc_len / avg_dl))
        score += idf * numerator / denominator

    return score

BM25 is the keyword half of hybrid retrieval. It excels at exact lexical matching — a query for “DSM-5 ADHD criteria” finds documents with those exact tokens, which dense embeddings might miss. The k1 parameter controls term frequency saturation (higher = more credit for repeated terms), and b controls length normalization (higher = more penalty for long documents).

    """

# Test Cases:
bm25 = ["doc_A", "doc_B", "doc_C", "doc_D"]

def reciprocal_rank_fusion(ranked_lists, k=60):
    scores = {}
    for rlist in ranked_lists:
        for rank, doc_id in enumerate(rlist, start=1):
            scores[doc_id] = scores.get(doc_id, 0) + 1.0 / (k + rank)
    return sorted(scores, key=lambda d: scores[d], reverse=True)

RRF is simple but remarkably effective. The constant k (typically 60) dampens the influence of high ranks, preventing any single system from dominating. A document ranked #1 by BM25 and #2 by vector search scores 1/61 + 1/62 ≈ 0.0326, beating one ranked #1 by only one system (0.0164). This is exactly how FirstThen’s hybrid retrieval combines BM25 keyword matching with FAISS semantic search.

import re

def parse_predicates(llm_output: str) -> list[dict]:
    results = []
    for line in llm_output.strip().split('\n'):
        line = line.strip()
        if not line or line.startswith('#'):
            continue

        match = re.match(r'(\w+)\((.*)\)$', line)
        if not match:
            continue

        name = match.group(1)
        args_str = match.group(2).strip()

        if not args_str:
            results.append({"name": name, "args": []})
            continue

        # Parse args respecting quoted strings
        args = []
        for arg in re.findall(r'"[^"]*"|\S[^,]*', args_str):
            arg = arg.strip().rstrip(',').strip()
            if arg.startswith('"') and arg.endswith('"'):
                args.append(arg[1:-1])
            elif arg.lstrip('-').isdigit():
                args.append(int(arg))
            else:
                args.append(arg)
        results.append({"name": name, "args": args})

    return results

This is the literal core task of the role. The LLM outputs predicates in a semi-structured format; this parser converts them into dicts that can be validated and fed to the ASP solver. In production, you’d add error recovery (partial parse attempts), logging for malformed output, and schema validation (see E16).

E6 Context Window Manager

Implement a context window manager that fits as many relevant retrieved chunks as possible into an LLM’s context window, prioritized by relevance score.

def build_context(
    chunks: list[dict],   # [{"text": str, "score": float, "source": str}]
    max_tokens: int,
    system_prompt_tokens: int,
    reserve_output_tokens: int

    """
    Select chunks to fit within the LLM context window.
    Available = max_tokens - system_prompt_tokens - reserve_output_tokens
    Sort by score desc, estimate tokens as len(text) // 4,
    greedily add until budget exhausted.
    Return selected chunks in relevance order.
    """
    pass

# Test Cases:
chunks = [
    {"text": "A" * 400, "score": 0.9, "source": "guide.pdf"},    # ~100 tokens
    {"text": "B" * 800, "score": 0.7, "source": "faq.pdf"},       # ~200 tokens
    {"text": "C" * 200, "score": 0.95, "source": "manual.pdf"},   # ~50 tokens
    {"text": "D" * 1200, "score": 0.6, "source": "ref.pdf"},      # ~300 tokens
]

assert len(selected) == 3
assert selected[0]["source"] == "manual.pdf"   # highest score
assert selected[1]["source"] == "guide.pdf"
assert build_context(chunks, 100, 50, 50) == []  # no room
def build_context(chunks, max_tokens, system_prompt_tokens, reserve_output_tokens):
    budget = max_tokens - system_prompt_tokens - reserve_output_tokens
    if budget <= 0:
        return []

    sorted_chunks = sorted(chunks, key=lambda c: c["score"], reverse=True)
    selected = []
    used = 0

    for chunk in sorted_chunks:
        est_tokens = len(chunk["text"]) // 4
        if used + est_tokens <= budget:
            selected.append(chunk)
            used += est_tokens

    return selected

Context window management is a critical RAG engineering skill. You always need to reserve space for the system prompt and expected output. Greedy by relevance is simple but effective — in practice you might also consider diversity (don’t pick 3 chunks from the same document) or recency weighting. The 4 chars/token estimate is rough; production systems use the model’s actual tokenizer (e.g., tiktoken).

def build_rag_prompt(user_query, context_chunks, extracted_predicates, system_rules):
    sections = []

    # 1. System instruction
    sections.append(
        "You are a parent coaching assistant for childhood ADHD.\n"
        "ONLY use information from the provided context.\n"
        "If the answer is not in the context, say "
        "'I don't have information about that.'\n"
        "NEVER provide medical advice, medication recommendations, or diagnoses."
    )

    # 2. Rules
    rules_text = "\n".join(f"- {r}" for r in system_rules)
    sections.append(f"## Rules\n{rules_text}")

    # 3. Context
    ctx_parts = []
    for chunk in context_chunks:
        ctx_parts.append(f"[Source: {chunk['source']}]\n{chunk['text']}")
    sections.append("## Context\n" + "\n\n".join(ctx_parts))

    # 4. Predicates
    pred_text = "\n".join(f"- {p}" for p in extracted_predicates)
    sections.append(f"## Current Session State\n{pred_text}")

    # 5. Query
    sections.append(f"## Parent Question\n{user_query}")

    return "\n\n".join(sections)

This prompt template enforces a strict contract: the LLM can ONLY use provided context, NEVER give medical advice, and always has visibility into the current session state from ASP. The section ordering matters — system instructions first anchors the LLM’s behavior before it sees any content. Source attribution enables traceability, which is important for clinical trust.

E8 Hallucination Detector

Given context chunks and an LLM response, detect claims NOT supported by the provided context. Critical safety mechanism for healthcare chatbots.

STOPWORDS = {"the", "a", "an", "is", "are", "was", "were", "be", "been",
    "being", "have", "has", "had", "do", "does", "did", "will", "would",
    "could", "should", "may", "might", "can", "shall", "to", "of", "in",
    "for", "on", "with", "at", "by", "from", "it", "this", "that", "and",

    "i", "my", "we", "our", "they", "their", "he", "she"}

def detect_unsupported_claims(
    response_sentences: list[str],
    context_chunks: list[str],
    threshold: float = 0.3
) -> list[dict]:
    """
    For each response sentence, compute word overlap with each chunk:
      overlap = |words_in_sentence ∩ words_in_chunk| / |words_in_sentence|
    (lowercase, alpha-only words, ignoring stopwords)
    Sentence is "supported" if max overlap >= threshold.
    Return [{"sentence": str, "best_overlap": float}] for unsupported only.
    """
    pass

# Test Cases:
context = [
    "Visual schedules help children with ADHD maintain morning routines",
    "Timers reduce transition anxiety for children aged 5 to 10",
]
response = [
    "Visual schedules can help your child with morning routines.",  # supported
    "Timers are effective for reducing transition anxiety.",         # supported
    "Medication such as Adderall should be considered.",            # NOT supported
    "Studies show 85% improvement rate with these techniques.",     # NOT supported

def detect_unsupported_claims(response_sentences, context_chunks, threshold=0.3):
    def get_words(text):
        return {w.lower() for w in text.split()
                if w.isalpha() and w.lower() not in STOPWORDS}

    chunk_words = [get_words(c) for c in context_chunks]
    unsupported = []

    for sent in response_sentences:
        sent_words = get_words(sent)
        if not sent_words:
            continue
        best = max(
            (len(sent_words & cw) / len(sent_words) for cw in chunk_words),
            default=0
        )
        if best < threshold:
            unsupported.append({"sentence": sent, "best_overlap": best})

    return unsupported

This is a simple word-overlap heuristic — production systems use NLI (natural language inference) models like cross-encoder/nli-deberta-v3-base to check entailment. But even this simple version catches obvious hallucinations like medication claims and fabricated statistics. In FirstThen’s healthcare context, ANY unsupported medical claim is dangerous. The ASP layer provides the first line of defense; this detector is the second.

E11. What is the key difference between “retrieval” and “reranking” in a RAG pipeline?

B is correct. Retrieval (BM25, bi-encoder) is fast but approximate — it finds top-K candidates. Reranking (cross-encoder) is slower but more accurate — it jointly encodes query + document to score relevance. This two-stage approach balances speed and quality.

C is correct. Prompting an LLM with “never recommend medication” is a suggestion it might violate. E13. What is “temperature” in LLM inference, and what setting would you use for predicate extraction?

B is correct. Temperature controls the softmax distribution over next tokens. Temperature ≈0 gives near-deterministic output, which is ideal for structured predicate extraction — the same input should produce the same predicates. For response generation, slightly higher temperature (0.3–0.7) produces more natural language.

A is correct. BM25 excels at exact lexical matching. “DSM-5 ADHD criteria” benefits from BM25 because the exact tokens match directly. Dense embeddings might map this to semantically similar but lexically different passages. FirstThen’s clinical terminology makes hybrid retrieval essential.

import math

def create_index(documents):
    return list(documents)

def query_index(index, query_embedding, top_k):
    def cosine_sim(a, b):
        dot = sum(x * y for x, y in zip(a, b))
        ma = math.sqrt(sum(x*x for x in a))
        mb = math.sqrt(sum(x*x for x in b))
        return dot / (ma * mb) if ma and mb else 0

    scored = [(doc["id"], cosine_sim(query_embedding, doc["embedding"]))
              for doc in index]
    scored.sort(key=lambda x: x[1], reverse=True)
    return [doc_id for doc_id, _ in scored[:top_k]]

This is brute-force O(n) search — exactly what FAISS’s IndexFlatIP does. For larger corpora, FAISS uses approximate methods (IVF, HNSW) to avoid scanning every vector. Understanding the exact algorithm helps debug retrieval quality issues — if top-K results look wrong, you can manually compute similarities to verify.

def validate_predicates(predicates, schema):
    valid, invalid = [], []

    for pred in predicates:
        name, args = pred["name"], pred["args"]

        if name not in schema:
            invalid.append({"predicate": pred, "error": f"Unknown predicate: {name}"})
            continue

        spec = schema[name]

        if len(args) != spec["arity"]:
            invalid.append({"predicate": pred,
                "error": f"Arity mismatch: expected {spec['arity']}, got {len(args)}"})
            continue

        arg_ok = True
        for i, (arg, atype) in enumerate(zip(args, spec["arg_types"])):
            constraint = spec["constraints"].get(str(i), {})

            if atype == "int":
                if not isinstance(arg, int):
                    invalid.append({"predicate": pred, "error": f"Arg {i}: expected int"})
                    arg_ok = False; break
                if "min" in constraint and arg < constraint["min"]:
                    invalid.append({"predicate": pred,
                        "error": f"Arg {i}: {arg} below min {constraint['min']}"})
                    arg_ok = False; break
                if "max" in constraint and arg > constraint["max"]:
                    invalid.append({"predicate": pred,
                        "error": f"Arg {i}: {arg} above max {constraint['max']}"})
                    arg_ok = False; break
            elif atype == "enum":
                if arg not in constraint.get("values", []):
                    invalid.append({"predicate": pred,
                        "error": f"Arg {i}: '{arg}' not in allowed values"})
                    arg_ok = False; break

        if arg_ok:
            valid.append(pred)

    return {"valid": valid, "invalid": invalid}

This validation layer sits between the LLM and the ASP solver. Without it, a hallucinated predicate like child_age(150) or challenge(flying) would enter the reasoning engine and produce nonsense. In healthcare, validation is defense-in-depth: even if the LLM makes a mistake, the system catches it before it affects reasoning.

def compress_conversation(turns, max_recent_turns, max_summary_tokens):
    recent = turns[-max_recent_turns:] if max_recent_turns < len(turns) else turns
    older = turns[:-max_recent_turns] if max_recent_turns < len(turns) else []

    facts_seen = set()
    facts = []
    for turn in older:
        for sentence in turn["text"].replace('. ', '.\n').split('\n'):
            sentence = sentence.strip()
            if not sentence:
                continue
            has_number = any(c.isdigit() for c in sentence)
            has_keyword = any(kw in sentence.lower() for kw in KEYWORDS)
            if has_number or has_keyword:
                key = sentence.lower()
                if key not in facts_seen:
                    facts_seen.add(key)
                    facts.append(sentence)

    summary = ". ".join(facts) if facts else ""
    # Trim to budget
    while len(summary) // 4 > max_summary_tokens and facts:
        facts.pop()
        summary = ". ".join(facts)

    total_tokens = len(summary) // 4
    for t in recent:
        total_tokens += len(t["text"]) // 4

    return {"summary": summary, "recent_turns": recent,
            "total_estimated_tokens": total_tokens}

Long coaching sessions can exceed context windows. This compressor preserves critical clinical facts (age, diagnosis, challenges) from early turns while keeping recent turns verbatim for conversational flow. Production systems might use an LLM to generate the summary, but keyword extraction is faster and doesn’t risk hallucinating new facts into the history.
